# 正 N 边形(可带圆角)路径数据生成器:核心/星芒都用它,方便调形状
# 注意:PowerShell 里 @(a, b) 中逗号的优先级比 + * 高,元素里带运算必须拆成变量,
#       否则 @($CX + Cos($a)*$R, $CY + ...) 会被解析成 $CX + (Cos($a)*($R,$CY))
function New-RoundPolyData {
    param(
        [double]$R, [int]$N, [double]$Round = 0.4, [double]$Phase = 0,
        [double]$CX = 120, [double]$CY = 122
    )
    $v = @()
    for ($i = 0; $i -lt $N; $i++) {
        $a = 2 * [Math]::PI * $i / $N + $Phase
        $px = $CX + [Math]::Cos($a) * $R
        $py = $CY + [Math]::Sin($a) * $R
        $v += , @($px, $py)
    }
    $k = $Round
    $en = @(); $ex = @()
    for ($i = 0; $i -lt $N; $i++) {
        $vi = $v[$i]
        $pv = $v[($i - 1 + $N) % $N]
        $nv = $v[($i + 1) % $N]
        $en1 = $vi[0] + ($pv[0] - $vi[0]) * $k
        $en2 = $vi[1] + ($pv[1] - $vi[1]) * $k
        $en += , @($en1, $en2)
        $ex1 = $vi[0] + ($nv[0] - $vi[0]) * $k
        $ex2 = $vi[1] + ($nv[1] - $vi[1]) * $k
        $ex += , @($ex1, $ex2)
    }
    $sb = New-Object System.Text.StringBuilder
    [void]$sb.Append('M ' + (F $ex[0][0]) + ',' + (F $ex[0][1]))
    for ($i = 1; $i -lt $N; $i++) {
        [void]$sb.Append(' L ' + (F $en[$i][0]) + ',' + (F $en[$i][1]))
        [void]$sb.Append(' Q ' + (F $v[$i][0]) + ',' + (F $v[$i][1]) + ' ' + (F $ex[$i][0]) + ',' + (F $ex[$i][1]))
    }
    [void]$sb.Append(' L ' + (F $en[0][0]) + ',' + (F $en[0][1]))
    [void]$sb.Append(' Q ' + (F $v[0][0]) + ',' + (F $v[0][1]) + ' ' + (F $ex[0][0]) + ',' + (F $ex[0][1]))
    [void]$sb.Append(' Z')
    return $sb.ToString()
}

# 核心:六边形(尖朝上),圆角较大,像一个软软的坐垫
# 回旋风车轮廓: 6 片连续曲面的叶片(内圈 40, 外尖 92, 每片扫过 36 度), 中间一块能放脸
function New-SwirlData {
    param([double]$CX = 120, [double]$CY = 122, [int]$N = 6, [double]$Ri = 40, [double]$Ro = 92, [double]$Sweep = 52)
    $sb = New-Object System.Text.StringBuilder
    for ($k = 0; $k -lt $N; $k++) {
        $a0 = 2 * [Math]::PI * $k / $N
        $a1 = $a0 + $Sweep * [Math]::PI / 180.0
        $a2 = 2 * [Math]::PI * ($k + 1) / $N
        $p0x = $CX + [Math]::Cos($a0) * $Ri; $p0y = $CY + [Math]::Sin($a0) * $Ri
        $p1x = $CX + [Math]::Cos($a1) * $Ro; $p1y = $CY + [Math]::Sin($a1) * $Ro
        $p2x = $CX + [Math]::Cos($a2) * $Ri; $p2y = $CY + [Math]::Sin($a2) * $Ri
        $m1 = $a0 + ($a1 - $a0) * 0.55
        $c1x = $CX + [Math]::Cos($m1) * $Ro * 0.60; $c1y = $CY + [Math]::Sin($m1) * $Ro * 0.70
        $m2 = $a1 + ($a2 - $a1) * 0.30
        $c2x = $CX + [Math]::Cos($m2) * $Ro * 0.88; $c2y = $CY + [Math]::Sin($m2) * $Ro * 0.80
        if ($k -eq 0) { [void]$sb.Append(('M {0:0.##},{1:0.##} ' -f $p0x, $p0y)) }
        [void]$sb.Append(('Q {0:0.##},{1:0.##} {2:0.##},{3:0.##} ' -f $c1x, $c1y, $p1x, $p1y))
        [void]$sb.Append(('Q {0:0.##},{1:0.##} {2:0.##},{3:0.##} ' -f $c2x, $c2y, $p2x, $p2y))
    }
    [void]$sb.Append('Z')
    return $sb.ToString()
}
$script:BodyData = New-SwirlData
